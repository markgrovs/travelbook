{{- $name := .Params.location | default .Params.place | default "" -}} {{- $lat := .Params.lat -}}{{- $long := .Params.long -}} {{- $hasLat := and (isset .Params "lat") (ne $lat nil) (ne (printf "%v" $lat) "") -}} {{- $hasLong := and (isset .Params "long") (ne $long nil) (ne (printf "%v" $long) "") -}} {{- $valid := false -}} {{- if ne $hasLat $hasLong -}}{{ errorf "Supply both lat and long in %s" .Path }}{{- end -}} {{- if and $hasLat $hasLong -}} {{- $numeric := `^[+-]?([0-9]+([.][0-9]*)?|[.][0-9]+)$` -}} {{- if not (and (findRE $numeric (printf "%v" $lat)) (findRE $numeric (printf "%v" $long))) -}} {{- errorf "lat and long must be numeric in %s" .Path -}} {{- else -}} {{- $lat = float $lat -}}{{- $long = float $long -}} {{- if or (lt $lat -90.0) (gt $lat 90.0) (lt $long -180.0) (gt $long 180.0) -}} {{- errorf "Coordinates out of range in %s" .Path -}} {{- else -}}{{- $valid = true -}}{{- end -}} {{- end -}} {{- end -}} {{- return (dict "name" $name "lat" $lat "long" $long "valid" $valid) -}}
